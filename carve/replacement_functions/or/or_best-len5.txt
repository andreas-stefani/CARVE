
static MachineBasicBlock* emit_or_best_len5(MachineInstr& MI, MachineBasicBlock* MBBOrig) {

MachineFunction* MF  = MBBOrig->getParent();
DebugLoc DL  = MI.getDebugLoc();
const TargetInstrInfo* TII = MF->getSubtarget().getInstrInfo();
MachineRegisterInfo& MRI = MF->getRegInfo();
Register RDest = MI.getOperand(0).getReg();
Register IN1 = MI.getOperand(1).getReg();
Register IN2 = MI.getOperand(2).getReg();


Register T1 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::AND), T1).addReg(IN1).addReg(IN2);
Register T2 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::SUB), T2).addReg(T1).addReg(IN2);
Register T3 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::XORI), T3).addReg(IN2).addImm(-1);
Register T4 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::XORI), T4).addReg(RISCV::X0).addImm(-1);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::SUB), RDest).addReg(IN1).addReg(T2);

MI.eraseFromParent();
return MBBOrig;
}
