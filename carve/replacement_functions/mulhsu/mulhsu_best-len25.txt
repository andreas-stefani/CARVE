BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::MUL), T8).addReg(T3).addReg(T6);
Register T9 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::MUL), T9).addReg(T4).addReg(T5);
