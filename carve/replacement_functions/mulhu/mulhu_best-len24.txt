
static MachineBasicBlock* emit_mulhu_best_len24(MachineInstr& MI, MachineBasicBlock* MBBOrig) {

MachineFunction* MF  = MBBOrig->getParent();
DebugLoc DL  = MI.getDebugLoc();
const TargetInstrInfo* TII = MF->getSubtarget().getInstrInfo();
MachineRegisterInfo& MRI = MF->getRegInfo();
Register RDest = MI.getOperand(0).getReg();
Register IN1 = MI.getOperand(1).getReg();
Register IN2 = MI.getOperand(2).getReg();


Register T1 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::LUI), T1).addImm(16);
Register T2 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::ADDI), T2).addReg(T1).addImm(-1);
Register T3 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::AND), T3).addReg(IN1).addReg(T2);
Register T4 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::SRLI), T4).addReg(IN1).addImm(16);
Register T5 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::AND), T5).addReg(T4).addReg(T2);
Register T6 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::AND), T6).addReg(IN2).addReg(T2);
Register T7 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::SRLI), T7).addReg(IN2).addImm(16);
Register T8 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::AND), T8).addReg(T7).addReg(T2);
Register T9 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::MUL), T9).addReg(T3).addReg(T6);
Register T10 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::MUL), T10).addReg(T3).addReg(T8);
Register T11 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::MUL), T11).addReg(T5).addReg(T6);
Register T12 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::MUL), T12).addReg(T5).addReg(T8);
Register T13 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::AND), T13).addReg(T10).addReg(T2);
Register T14 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::AND), T14).addReg(T11).addReg(T2);
Register T15 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::SRLI), T15).addReg(T9).addImm(16);
Register T16 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::ADD), T16).addReg(T13).addReg(T14);
Register T17 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::ADD), T17).addReg(T16).addReg(T15);
Register T18 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::SRLI), T18).addReg(T17).addImm(16);
Register T19 = MRI.createVirtualRegister(&RISCV::GPRRegClass);

BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::SRLI), T20).addReg(T11).addImm(16);
Register T21 = MRI.createVirtualRegister(&RISCV::GPRRegClass);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::ADD), T21).addReg(T12).addReg(T19);

BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::ADD), T22).addReg(T21).addReg(T20);
BuildMI(*MBBOrig, MI, DL, TII->get(RISCV::ADD), RDest).addReg(T22).addReg(T18);

MI.eraseFromParent();
return MBBOrig;
}
